local function FullPath(obj)
    if not obj or not obj.Parent then return obj and obj.Name or "nil" end
    local parts, cur = {}, obj
    while cur and cur ~= game do
        table.insert(parts, 1, cur.Name)
        cur = cur.Parent
    end
    return table.concat(parts, ".")
end

local function SafeGet(obj, prop)
    local ok, v = pcall(function() return obj[prop] end)
    return ok and v or nil
end

local function IsScript(obj)
    return obj:IsA("Script") or obj:IsA("LocalScript") or obj:IsA("ModuleScript")
end

local function ScriptSource(obj)
    if not IsScript(obj) then return nil end
    local ok, src = pcall(function() return obj.Source end)
    if ok and src and #src > 0 then return src end
    return nil
end

local function SearchSource(src, patterns)
    if not src then return {} end
    local hits = {}
    for _, pat in ipairs(patterns) do
        local lineNum = 0
        for line in src:gmatch("[^\n]+") do
            lineNum = lineNum + 1
            if line:lower():find(pat:lower(), 1, true) then
                table.insert(hits, { pattern = pat, line = lineNum, text = line:gsub("^%s+", ""):sub(1, 120) })
            end
        end
    end
    return hits
end

local function ShortSource(src, maxLines)
    if not src then return "(unavailable)" end
    maxLines = maxLines or 40
    local lines, n = {}, 0
    for line in src:gmatch("[^\n]+") do
        n = n + 1
        if n > maxLines then table.insert(lines, ("...+%d linhas"):format(n)); break end
        table.insert(lines, line)
    end
    return table.concat(lines, "\n")
end

local function CountLines(src)
    if not src then return 0 end
    return select(2, src:gsub("\n","")) + 1
end

local function MapObject(obj, depth)
    depth = depth or 0
    local entry = {
        name = obj.Name, class = obj.ClassName,
        path = FullPath(obj), depth = depth,
        children_count = 0, scripts = {}, properties = {},
    }
    local relevantProps = {
        BasePart   = {"Size","Position","Anchored","CanCollide"},
        Humanoid   = {"WalkSpeed","JumpPower","AutoRotate","Health","MaxHealth"},
        Camera     = {"CameraType","CameraSubject","FieldOfView"},
        Tool       = {"RequiresHandle","ToolTip","GripForward","GripPos"},
        RemoteEvent= {}, RemoteFunction={},
        Sound      = {"SoundId","Volume","Looped"},
        Animation  = {"AnimationId"},
    }
    for base, props in pairs(relevantProps) do
        if obj:IsA(base) then
            for _, p in ipairs(props) do
                local v = SafeGet(obj, p)
                if v ~= nil then entry.properties[p] = tostring(v) end
            end
        end
    end
    if IsScript(obj) then
        local src = ScriptSource(obj)
        table.insert(entry.scripts, { name=obj.Name, class=obj.ClassName, path=FullPath(obj), lines=CountLines(src) })
        table.insert(Report.K_scripts, { name=obj.Name, class=obj.ClassName, path=FullPath(obj), lines=CountLines(src) })
    end
    if depth < 7 then
        for _, child in ipairs(obj:GetChildren()) do
            entry.children_count = entry.children_count + 1
            if depth < 4 then
                table.insert(entry, MapObject(child, depth+1))
            end
        end
    end
    return entry
end

local function RunA()

local function AnalyzeWeapon(tool)
    local entry = {
        name=tool.Name, path=FullPath(tool),
        handle_found = tool:FindFirstChild("Handle") ~= nil,
        grip = {
            GripForward = SafeGet(tool,"GripForward") and tostring(SafeGet(tool,"GripForward")),
            GripPos     = SafeGet(tool,"GripPos")     and tostring(SafeGet(tool,"GripPos")),
        },
        scripts={}, activation_flow={}, aim_evidence={},
        anim_evidence={}, cooldown_evidence={}, confidence="HIPÓTESE",
    }
    for _, obj in ipairs(tool:GetDescendants()) do
        if IsScript(obj) then
            local src = ScriptSource(obj)
            local se = { name=obj.Name, class=obj.ClassName, path=FullPath(obj), lines=CountLines(src) }
            if src then
                for _,h in ipairs(SearchSource(src,{"Activated","MouseButton1","Equipped","Unequipped"})) do
                    table.insert(entry.activation_flow,{script=obj.Name,line=h.line,text=h.text})
                end
                for _,h in ipairs(SearchSource(src,{"Mouse.Hit","UnitRay","ScreenPointToRay","ViewportPointToRay","Raycast","LookVector","lookAt"})) do
                    table.insert(entry.aim_evidence,{script=obj.Name,line=h.line,text=h.text})
                end
                for _,h in ipairs(SearchSource(src,{"LoadAnimation","AnimationTrack",":Play()","AdjustSpeed"})) do
                    table.insert(entry.anim_evidence,{script=obj.Name,line=h.line,text=h.text})
                end
                for _,h in ipairs(SearchSource(src,{"debounce","cooldown","tick()","os.clock","task.wait","wait("})) do
                    table.insert(entry.cooldown_evidence,{script=obj.Name,line=h.line,text=h.text})
                end
                entry.confidence = #entry.activation_flow > 0 and "CONFIRMADO" or "PROVÁVEL"
                se.preview = ShortSource(src, 30)
            else
                entry.confidence = "DESCONHECIDO"
            end
            table.insert(entry.scripts, se)
        end
    end
    return entry
end

local function RunB()

local function RunI()
    Report.I_remotes = {}
    local function FindRemotes(root)
        for _, obj in ipairs(root:GetDescendants()) do
            if obj:IsA("RemoteEvent") or obj:IsA("RemoteFunction") or obj:IsA("BindableEvent") then
                local entry={
                    name=obj.Name, class=obj.ClassName, path=FullPath(obj),
                    callers={}, apparent_function="DESCONHECIDO",
                    server_handler="DESCONHECIDO", validations={},
                }
                for _, sroot in ipairs(ROOTS) do
                    for _, sobj in ipairs(sroot:GetDescendants()) do
                        if IsScript(sobj) then
                            local src = ScriptSource(sobj)
                            if src and src:find(obj.Name,1,true) then
                                for _,h in ipairs(SearchSource(src,{
                                    obj.Name,"FireServer","FireClient","FireAllClients",
                                    "InvokeServer","InvokeClient",
                                    "OnServerEvent","OnClientEvent","OnServerInvoke","OnClientInvoke",
                                })) do
                                    table.insert(entry.callers,{script=FullPath(sobj),line=h.line,text=h.text,pattern=h.pattern})
                                    if h.pattern:find("FireServer")     then entry.apparent_function="Cliente → Servidor" end
                                    if h.pattern:find("OnServerEvent") or h.pattern:find("OnServerInvoke") then
                                        entry.server_handler=FullPath(sobj).." L"..h.line
                                        entry.apparent_function="Recebido no servidor"
                                    end
                                    if h.pattern:find("FireAllClients") or h.pattern:find("FireClient") then
                                        entry.apparent_function="Servidor → Cliente(s)"
                                    end
                                end
                            end
                        end
                    end
                end
                table.insert(Report.I_remotes, entry)
            end
        end
    end
    for _, r in ipairs(ROOTS) do FindRemotes(r) end
end

local function RunJ()
    Report.J_security = {}
    for _, root in ipairs(ROOTS) do
        for _, obj in ipairs(root:GetDescendants()) do
            if IsScript(obj) then
                local src = ScriptSource(obj)
                if not src then goto continue end
                local fireLines  = {}; for _,h in ipairs(SearchSource(src,{"FireServer","InvokeServer"})) do table.insert(fireLines,h.line) end
                local distLines  = {}; for _,h in ipairs(SearchSource(src,{"Magnitude","Distance","maxDistance"})) do table.insert(distLines,h.line) end
                local cdLines    = {}; for _,h in ipairs(SearchSource(src,{"debounce","cooldown","tick()","os.clock"})) do table.insert(cdLines,h.line) end
                local createLines= {}; for _,h in ipairs(SearchSource(src,{"Instance.new",":Clone()"})) do table.insert(createLines,h.line) end
                local lsLines    = {}; for _,h in ipairs(SearchSource(src,{"loadstring"})) do table.insert(lsLines,h.line) end
                local authLines  = {}; for _,h in ipairs(SearchSource(src,{"GetPlayerFromCharacter","assert(","error(","FindFirstChild.*Players"})) do table.insert(authLines,h.line) end

                if #fireLines > 0 and #cdLines == 0 then
                    table.insert(Report.J_security,{
                        file=FullPath(obj), severity="MÉDIO", status="PROVÁVEL",
                        issue="FireServer/InvokeServer sem cooldown visível",
                        impact="Spam de eventos, possível abuso de mecânicas",
                        evidence="FireServer nas linhas: "..table.concat(fireLines,", "),
                        fix="Adicionar debounce ou cooldown server-side antes de processar",
                    })
                end
                if #fireLines > 0 and #distLines == 0 then
                    table.insert(Report.J_security,{
                        file=FullPath(obj), severity="MÉDIO", status="PROVÁVEL",
                        issue="FireServer sem verificação de distância",
                        impact="Jogador pode acionar eventos a qualquer distância",
                        evidence="FireServer nas linhas: "..table.concat(fireLines,", ").."; sem Magnitude",
                        fix="Validar distância no servidor antes de processar",
                    })
                end
                if #lsLines > 0 then
                    table.insert(Report.J_security,{
                        file=FullPath(obj), severity="ALTO", status="CONFIRMADO",
                        issue="loadstring() encontrado",
                        impact="Execução de código dinâmico — alto risco",
                        evidence="Linhas: "..table.concat(lsLines,", "),
                        fix="Remover loadstring; usar módulos estáticos",
                    })
                end
                if obj:IsA("Script") and #createLines > 0 and #authLines == 0 then
                    table.insert(Report.J_security,{
                        file=FullPath(obj), severity="BAIXO", status="HIPÓTESE",
                        issue="Criação de Instance sem verificação de autorização visível",
                        impact="Possível criação de objetos não autorizada",
                        evidence="Instance.new/:Clone() nas linhas: "..table.concat(createLines,", "),
                        fix="Validar player autorizado antes de criar objetos",
                    })
                end
                ::continue::
            end
        end
    end
end

-- EXPORTAÇÃO — gera texto completo copiável
-- ============================================================
local function BuildTextReport()
    local L = {}
    local function W(s) table.insert(L, tostring(s or "")) end
    local function HR() W(("─"):rep(70)) end
    local function H1(t) W("") HR() W("  ■ "..t) HR() end
    local function H2(t) W("  ┌─ "..t) end
    local function KV(k,v) W("    "..k.." : "..tostring(v)) end

    W("╔══════════════════════════════════════════════════════════════════════╗")
    W("║         DEEP INSPECTOR — RELATÓRIO TÉCNICO COMPLETO                 ║")
    W("╚══════════════════════════════════════════════════════════════════════╝")
    W("  Gerado   : "..Report.metadata.generated_at)
    W("  Versão   : "..Report.metadata.version)
    W("  Executor : Delta (client-side)")

    -- A
    H1("A. ESTRUTURA DO JOGO")
    for _, node in ipairs(Report.A_structure) do
        H2(node.path.." ["..node.class.."]")
        KV("Filhos diretos", node.children_count)
        if next(node.properties) then
            W("    Propriedades:")
            for k,v in pairs(node.properties) do W("      "..k.." = "..v) end
        end
        if #node.scripts > 0 then
            W("    Scripts:")
            for _, s in ipairs(node.scripts) do W("      "..s.class.." "..s.name.." ("..s.lines.." linhas)") end
        end
    end

    -- B
    H1("B. SISTEMA DE ARMAS")
    if #Report.B_weapons == 0 then W("  Nenhuma Tool encontrada nas raízes acessíveis.") end
    for _, w in ipairs(Report.B_weapons) do
        H2(w.name.." — "..w.confidence)
        KV("Caminho", w.path)
        KV("Handle", w.handle_found)
        KV("GripForward", w.grip.GripForward or "N/A")
        KV("GripPos", w.grip.GripPos or "N/A")
        KV("Scripts", #w.scripts)
        KV("Evidências ativação", #w.activation_flow)
        KV("Evidências mira", #w.aim_evidence)
        KV("Evidências animação", #w.anim_evidence)
        KV("Evidências cooldown", #w.cooldown_evidence)
        if #w.activation_flow > 0 then
            W("    Fluxo de ativação:")
            for _, ev in ipairs(w.activation_flow) do W("      L"..ev.line.." ["..ev.script.."] "..ev.text) end
        end
        if #w.aim_evidence > 0 then
            W("    Evidências de mira:")
            for _, ev in ipairs(w.aim_evidence) do W("      L"..ev.line.." ["..ev.script.."] "..ev.text) end
        end
    end

    -- C
    H1("C. SISTEMA DE MIRA")
    for field, data in pairs(Report.C_aim) do
        H2(field.." ["..data.status.."]")
        for _, ev in ipairs(data.evidence) do W("    L"..ev.line.." ["..ev.script.."] "..ev.text) end
        if #data.evidence == 0 then W("    (sem evidência encontrada)") end
    end

    -- D
    H1("D. CÂMERA")
    H2("Estado atual")
    for k,v in pairs(Report.D_camera.current_state) do KV(k,v) end
    H2("Evidências em scripts ("..#Report.D_camera.evidence..")")
    for _, ev in ipairs(Report.D_camera.evidence) do
        W("    L"..ev.line.." ["..ev.script.."] "..ev.text)
    end

    -- E
    H1("E. SHIFT LOCK")
    H2("Configurações engine")
    for k,v in pairs(Report.E_shiftlock.engine_setting) do KV(k,v) end
    H2("Tabela de comportamento")
    W("    "..("─"):rep(66))
    W(string.format("    %-28s | %-14s | %-14s | %s","Aspecto","ShiftLock ON","ShiftLock OFF","Status"))

    for _, row in ipairs(Report.E_shiftlock.behavior_table) do

    end
    W("    "..("─"):rep(66))
    H2("Evidências em scripts ("..#Report.E_shiftlock.evidence..")")
    for _, ev in ipairs(Report.E_shiftlock.evidence) do
        W("    L"..ev.line.." ["..ev.script.."] "..ev.text)
    end

    -- F
    H1("F. ROTAÇÃO DO PERSONAGEM")
    KV("Humanoid.AutoRotate", Report.F_rotation.humanoid_autorotate)
    H2("Síntese")
    for k,v in pairs(Report.F_rotation.summary) do
        W("    "..k.." : encontrado="..tostring(v.found).." ["..v.status.."]")
    end
    H2("Evidências ("..#Report.F_rotation.evidence..")")
    for _, ev in ipairs(Report.F_rotation.evidence) do
        W("    L"..ev.line.." ["..ev.script.."] "..ev.text)
    end

    -- G
    H1("G. SISTEMA DE PORTAIS — confiança: "..Report.G_portals.confidence)
    KV("Scripts relevantes encontrados", #Report.G_portals.scripts_found)
    KV("Evidências criação", #Report.G_portals.creation_flow)
    KV("Evidências posicionamento", #Report.G_portals.position_calc)
    KV("Evidências orientação", #Report.G_portals.orientation_calc)
    KV("Evidências superfície", #Report.G_portals.surface_selection)
    KV("Evidências modelo", #Report.G_portals.model_creation)
    KV("Evidências linkage", #Report.G_portals.portal_linkage)
    KV("Evidências teleporte", #Report.G_portals.teleport_execution)
    if #Report.G_portals.scripts_found > 0 then
        H2("Scripts relacionados a portais")
        for _, se in ipairs(Report.G_portals.scripts_found) do
            W("    "..se.class.." : "..se.path.." ("..#se.hits.." hits)")
            for _, h in ipairs(se.hits) do W("      L"..h.line.." ["..h.pattern.."] "..h.text) end
        end
    end
    if #Report.G_portals.creation_flow > 0 then
        H2("Fluxo de criação")
        for _, ev in ipairs(Report.G_portals.creation_flow) do W("    L"..ev.line.." ["..ev.script.."] "..ev.text) end
    end
    if #Report.G_portals.teleport_execution > 0 then
        H2("Execução do teleporte")
        for _, ev in ipairs(Report.G_portals.teleport_execution) do W("    L"..ev.line.." ["..ev.script.."] "..ev.text) end
    end

    -- H
    H1("H. TELEPORTE — confiança: "..Report.H_teleport.confidence)
    KV("Evidências", #Report.H_teleport.evidence)
    for _, ev in ipairs(Report.H_teleport.evidence) do
        W("    L"..ev.line.." ["..ev.script.."] "..ev.text)
    end

    -- I
    H1("I. COMUNICAÇÃO CLIENTE-SERVIDOR")
    if #Report.I_remotes == 0 then W("  Nenhum RemoteEvent/RemoteFunction encontrado nas raízes acessíveis.") end
    for _, rem in ipairs(Report.I_remotes) do
        H2(rem.name.." ("..rem.class..")")
        KV("Caminho", rem.path)
        KV("Função aparente", rem.apparent_function)
        KV("Handler servidor", rem.server_handler)
        KV("Callers encontrados", #rem.callers)
        for _, c in ipairs(rem.callers) do
            W("    L"..c.line.." ["..c.pattern.."] "..c.script)
        end
    end

    -- J
    H1("J. AUDITORIA DE SEGURANÇA")
    if #Report.J_security == 0 then W("  Nenhum problema encontrado nas raízes acessíveis.") end
    for i, f in ipairs(Report.J_security) do
        H2("["..i.."] "..f.severity.." — "..f.issue.." ["..f.status.."]")
        KV("Arquivo", f.file)
        KV("Impacto", f.impact)
        KV("Evidência", f.evidence)
        KV("Correção", f.fix)
    end

    -- K
    H1("K. SCRIPTS MAPEADOS — total: "..#Report.K_scripts)
    for _, s in ipairs(Report.K_scripts) do
        W("  "..s.class.." | "..s.path.." | "..s.lines.." linhas")
    end

    -- L
    H1("L. INFORMAÇÕES NÃO VERIFICADAS")
    for _, note in ipairs(Report.L_unverified) do W("  • "..note) end

    HR()
    W("  FIM DO RELATÓRIO")
    HR()

    return table.concat(L, "\n")
end

local Monitor = { active=false, connections={}, log={} }

local function MonitorLog(cat, msg, data)
    table.insert(Monitor.log, {time=tick(), category=cat, message=msg, data=data or {}})
end

local function StartMonitor()
    if Monitor.active then return end
    Monitor.active = true
    for _, root in ipairs({Workspace, ReplicatedStorage or Workspace}) do
        if root then
            local c1 = root.DescendantAdded:Connect(function(o)
                MonitorLog("ADDED",   o.Name.." → "..FullPath(root), {class=o.ClassName,path=FullPath(o)})
            end)
            local c2 = root.DescendantRemoving:Connect(function(o)
                MonitorLog("REMOVED", o.Name.." ← "..FullPath(root), {class=o.ClassName})
            end)
            table.insert(Monitor.connections,c1)
            table.insert(Monitor.connections,c2)
        end
    end
    if LocalPlayer then
        local function WatchPlayer(p)
            p.CharacterAdded:Connect(function(char)
                MonitorLog("CHAR_ADDED", p.Name, {player=p.Name})
                char.ChildAdded:Connect(function(ch)
                    if ch:IsA("Tool") then MonitorLog("TOOL_EQUIPPED",p.Name.." → "..ch.Name,{}) end
                end)
                char.ChildRemoved:Connect(function(ch)
                    if ch:IsA("Tool") then MonitorLog("TOOL_REMOVED",p.Name.." ← "..ch.Name,{}) end
                end)
                local root = char:FindFirstChild("HumanoidRootPart")
                if root then
                    local lastPos = root.Position
                    local hc = RunService.Heartbeat:Connect(function()
                        if not Monitor.active then return end
                        local delta = (root.Position - lastPos).Magnitude
                        if delta > 10 then
                            MonitorLog("LARGE_MOVE",p.Name.." +"..string.format("%.1f",delta).." studs",{from=tostring(lastPos),to=tostring(root.Position)})
                        end
                        lastPos = root.Position
                    end)
                    table.insert(Monitor.connections, hc)
                end
            end)
        end
        for _,p in ipairs(Players:GetPlayers()) do WatchPlayer(p) end
        local pac = Players.PlayerAdded:Connect(WatchPlayer)
        table.insert(Monitor.connections, pac)
    end
    -- monitora portais no Workspace
    local pc = Workspace.DescendantAdded:Connect(function(o)
        if o.Name:lower():find("portal") then
            MonitorLog("PORTAL_DETECTED", o.Name, {class=o.ClassName,path=FullPath(o)})
        end
    end)
    table.insert(Monitor.connections, pc)
end

local function StopMonitor()
    Monitor.active = false
    for _, c in ipairs(Monitor.connections) do pcall(function() c:Disconnect() end) end
    Monitor.connections = {}
end
